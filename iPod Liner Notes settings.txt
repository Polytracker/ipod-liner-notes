# Settings for iPod Liner Notes.
# Change a value after the = sign and save. A line starting with # is ignored.

# The iPod's drive letter, as File Explorer shows it.
ipod drive = G:

# The music folder on the iPod. Leave empty for the Music folder on the drive.
music folder =

# Artist bios, for PodBox's Artist Bio screen, and liner notes, for its Liner
# Notes screen: auto writes them only if the PodBox on the iPod has that
# screen, yes always, no never.
bios = auto
liner notes = auto

# yes or no: look again, on every run, for artists with no bio and albums
# with no notes. Albums and artists that already have them are left alone.
recheck missing = yes

# Days before an artist's bio is checked again (0 = never). Artists who have
# died and bands that have split are never rechecked. A bio is only replaced
# if it has really changed, and never if you edited it yourself.
bio recheck days = 365

# Days before an album's liner notes are checked again (0 = never). As with
# bios, they're only replaced if they've really changed and you haven't
# edited them.
notes recheck days = 365

# yes or no: make the banner pictures at the top of the Artist Bio and Liner
# Notes screens. no writes the text only. The artist photos for the rest of
# the iPod are fetched either way.
banners = yes
